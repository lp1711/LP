ML: Gradient Descent

def f(x):
    return (x + 3)**2

def df(x):
    return 2 * (x + 3)

x = 2
learning_rate = 0.1
iterations = 100

for i in range(iterations):
    x = x - learning_rate * df(x)

print("Local Minima at x =", x)
print("Minimum value of y =", f(x))

BT: Library management
pragma solidity ^0.8.0;

contract Library {

    struct Book {
        uint id;
        string title;
        string author;
    }

    Book[] public books;

    function addBook(uint _id, string memory _title, string memory _author) public {
        books.push(Book(_id, _title, _author));
    }

    function getBook(uint index) public view returns (uint, string memory, string memory) {
        Book memory b = books[index];
        return (b.id, b.title, b.author);
    }

    function getCount() public view returns (uint) {
        return books.length;
    }
}
